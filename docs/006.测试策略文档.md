##### 3.2 营养计算服务单元测试（TDEE/BMR）

**测试类**：`NutritionServiceTest`

**覆盖的算法：**

| 算法 | 公式 | 测试用例数 |
|------|------|-----------|
| BMR（男性） | `10W + 6.25H - 5A + 5` | ≥ 10 |
| BMR（女性） | `10W + 6.25H - 5A - 161` | ≥ 10 |
| TDEE | `BMR × 活动系数` | ≥ 8 |
| 饮食模式系数 | weight_loss=0.80 / muscle_gain=1.10 等 | ≥ 6 |

void NutritionServiceTest::bmrBoundaryTest() {
    QFETCH(int, age);
    QFETCH(double, height);
    QFETCH(double, weight);
    QFETCH(QString, gender);
    QFETCH(double, expectedBmr);

    double result = NutritionService::calculateBMR(gender, height, weight, age);
    if (expectedBmr < 0) {
        // 异常值应返回 -1 或抛出异常
        QVERIFY(result < 0 || result == -1);
    } else {
        // 允许 ±0.01 浮点误差

void InventoryBatchTest::testFIFODeduction() {
    // 按 purchase_date ASC, expiry_date ASC 排序后扣减
    auto sorted = batchService.sortByFIFO(initialBatches);
    auto deductions = batchService.deduct(sorted, consumeAmount);

    // 验证扣减顺序
    QVERIFY(deductions[0].batchId == sorted[0].id); // 最早的先扣
    // 验证扣减量不超过批次库存
    for (const auto& d : deductions) {
        QVERIFY(d.deductQuantity <= d.availableQuantity);
    }
    // 验证总量匹配
    double totalDeducted = 0;
    for (const auto& d : deductions) totalDeducted += d.deductQuantity;
    QFUZZYCOMPARE(totalDeducted, consumeAmount, 0.01);
}

void SyncEngineTest::testConflictResolution() {
    auto resolution = syncEngine.resolveConflict(localEntity, serverEntity);
    if (localEntity.updatedAt > serverEntity.updatedAt) {
        QVERIFY(resolution == ExpectedAction::PushLocal);
    } else if (localEntity.updatedAt < serverEntity.updatedAt) {

void WebSocketSyncTest::testInventoryPushWithin2Seconds() {
    // 1. 客户端连接
    wsClient.connect(wsUrl, token);

    // 2. 客户端提交入库操作
    inventoryService.addBatch(batchData);

void WebSocketSyncTest::testReconnectAfterDisconnect() {

    wsClient.connect(wsUrl, token);
    QVERIFY(wsClient.isConnected());

void MultiDeviceTest::testConcurrentInventoryDeduct() {
    // 场景：设备A和设备B同时对同一批次进行核销
    // 初始：番茄库存 500g（单个批次）

    // 设备A准备：标记"番茄炒蛋"完成（需消耗300g）
    clientA.markCooking(menuIdA);
    auto deductA = clientA.getDeductionPreview(menuIdA);
    QVERIFY(deductA.totalConsume == 300);

    // 设备B准备：标记"番茄蛋花汤"完成（需消耗200g）
    clientB.markCooking(menuIdB);
    auto deductB = clientB.getDeductionPreview(menuIdB);
    QVERIFY(deductB.totalConsume == 200);

    // 并发执行核销
    QtConcurrent::run([&] { clientA.completeCooking(menuIdA, deductA); });
    QtConcurrent::run([&] { clientB.completeCooking(menuIdB, deductB); });

    // 等待结果（乐观锁冲突处理）
    QTRY_VERIFY(clientA.isOperationCompleted() || clientA.hasConflict());
    QTRY_VERIFY(clientB.isOperationCompleted() || clientB.hasConflict());

    // 验证：至少一个成功，另一个收到冲突通知
    bool aSuccess = clientA.isOperationCompleted();
    bool bSuccess = clientB.isOperationCompleted();
    QVERIFY(aSuccess != bSuccess); // 不能同时成功（总库存500g，只能满足一个）

    // 验证最终库存一致（500 - 300 = 200 或 500 - 200 = 300）
    auto finalStock = inventoryService.getTotalStock(ingredientId);
    QVERIFY(finalStock == 200 || finalStock == 300);
}

void MultiDeviceTest::testOfflineThenOnlineSync() {
    // 场景：设备A离线时修改，设备B在线修改，重连后冲突解决

    // Step 1: 设备A进入离线模式
    clientA.enterOfflineMode();

    // Step 2: 设备A离线修改菜单
    clientA.updateMenu(menuIdA, { diet_mode: "cheat_day" });

    // Step 3: 设备B在线修改同一菜单
    clientB.updateMenu(menuIdA, { diner_count: 5 });
    QVERIFY(clientB.isOperationSuccessful());

    // Step 4: 设备A恢复网络
    clientA.exitOfflineMode();

    // Step 5: 验证设备A推送离线变更到 sync_queue
    auto queueItems = clientA.getSyncQueue();
    QVERIFY(queueItems.count() >= 1);

    // Step 6: 验证冲突解决（时间戳较新的覆盖）
    auto resolved = clientA.getResolvedMenu(menuIdA);
    // 设备B的操作时间更晚，应保留设备B的变更或进行合并
    QVERIFY(resolved.diner_count == 5);
}

```cpp
class SyncLatencyBenchmark {
public:
    void runBenchmark(int iterations = 100) {
        QVector<qint64> latencies;

        for (int i = 0; i < iterations; ++i) {
            // 准备测试数据
            auto batch = createTestBatch();

            // 记录操作时间戳
            qint64 opTimestamp = QDateTime::currentMSecsSinceEpoch();

            // 执行操作
            inventoryService.addBatch(batch);

            // Mock Server 记录收到时间
            qint64 serverTimestamp = mockServer.waitForMessage(
                MessageType::InventoryChanged, 3000
            ).timestamp;

            qint64 latency = serverTimestamp - batch.updatedAt;
            latencies.append(latency);
        }

        // 计算统计指标
        std::sort(latencies.begin(), latencies.end());
        qint64 p50 = latencies[iterations * 0.50];
        qint64 p90 = latencies[iterations * 0.90];
        qint64 p99 = latencies[iterations * 0.99];
        qint64 avg = std::accumulate(latencies) / iterations;

        qDebug() << QString("同步延迟: 平均=%1ms P50=%2ms P90=%3ms P99=%4ms")
                        .arg(avg).arg(p50).arg(p90).arg(p99);

        // 断言 P99 ≤ 2000ms
        QVERIFY2(p99 <= 2000,
            qPrintable(QString("P99 延迟 %1ms 超过 2000ms 目标").arg(p99)));
        QVERIFY2(avg <= 500,
            qPrintable(QString("平均延迟 %1ms 超过 500ms 目标").arg(avg)));
    }
};
```

| 多设备同步 | 库存1条变更 | 3设备 | ≤ 1000ms |
